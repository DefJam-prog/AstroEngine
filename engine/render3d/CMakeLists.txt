if(ASTRO_BUILD_GLTF_IMPORTER)
    set(_astro_fastgltf_source "${ASTRO_GLTF_DEPS_ROOT}/fastgltf-v1.0.1")
    set(_astro_simdjson_header "${ASTRO_GLTF_DEPS_ROOT}/simdjson/simdjson.h")
    set(_astro_simdjson_source "${ASTRO_GLTF_DEPS_ROOT}/simdjson/simdjson.cpp")
    set(_astro_simdutf_header "${ASTRO_GLTF_DEPS_ROOT}/simdutf/simdutf.h")
    set(_astro_simdutf_source "${ASTRO_GLTF_DEPS_ROOT}/simdutf/simdutf.cpp")
    foreach(_astro_required_file IN ITEMS
            "${_astro_fastgltf_source}/CMakeLists.txt"
            "${_astro_simdjson_header}" "${_astro_simdjson_source}"
            "${_astro_simdutf_header}" "${_astro_simdutf_source}")
        if(NOT EXISTS "${_astro_required_file}")
            message(FATAL_ERROR "ASTRO_BUILD_GLTF_IMPORTER requires locally staged dependency file: ${_astro_required_file}")
        endif()
    endforeach()

    find_package(Git REQUIRED)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${_astro_fastgltf_source}" rev-parse HEAD
        OUTPUT_VARIABLE _astro_fastgltf_commit OUTPUT_STRIP_TRAILING_WHITESPACE
        RESULT_VARIABLE _astro_fastgltf_git_result)
    if(NOT _astro_fastgltf_git_result EQUAL 0 OR NOT _astro_fastgltf_commit STREQUAL "87cfbceeeda127ccc9092694279437b700046119")
        message(FATAL_ERROR "Expected fastgltf v1.0.1 at commit 87cfbceeeda127ccc9092694279437b700046119")
    endif()

    function(_astro_verify_sha256 path expected)
        file(SHA256 "${path}" actual)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "SHA-256 mismatch for pinned glTF dependency: ${path}")
        endif()
    endfunction()
    _astro_verify_sha256("${_astro_simdjson_header}" "ca53ddca5f927b993ded4225b8fd85f18f9374e70e4f2d56376f42d661ff010a")
    _astro_verify_sha256("${_astro_simdjson_source}" "0652d8dd436b210a8840267cbd0bb4c1908f6551790e7dcd543412ba54b002c9")
    _astro_verify_sha256("${_astro_simdutf_header}" "90d39faf26030d18a6caec00b34a09ca4855804e7c2a9d59bf93d5d506347486")
    _astro_verify_sha256("${_astro_simdutf_source}" "566033aa28edfaaa12aff8dfcfc720bc0dea0af1c0a0462bbd903c4cba1814b7")

    add_library(astro_gltf_simdjson STATIC "${_astro_simdjson_source}")
    target_include_directories(astro_gltf_simdjson PUBLIC "${ASTRO_GLTF_DEPS_ROOT}/simdjson")
    target_compile_features(astro_gltf_simdjson PUBLIC cxx_std_20)
    add_library(simdjson::simdjson ALIAS astro_gltf_simdjson)

    add_library(astro_gltf_simdutf STATIC "${_astro_simdutf_source}")
    target_include_directories(astro_gltf_simdutf PUBLIC "${ASTRO_GLTF_DEPS_ROOT}/simdutf")
    target_compile_features(astro_gltf_simdutf PUBLIC cxx_std_20)
    target_compile_definitions(astro_gltf_simdutf PUBLIC
        SIMDUTF_FEATURE_BASE64=1 SIMDUTF_FEATURE_DETECT_ENCODING=0 SIMDUTF_FEATURE_ASCII=0
        SIMDUTF_FEATURE_LATIN1=0 SIMDUTF_FEATURE_UTF8=0 SIMDUTF_FEATURE_UTF16=0 SIMDUTF_FEATURE_UTF32=0)
    add_library(simdutf::simdutf ALIAS astro_gltf_simdutf)

    set(FASTGLTF_DOWNLOAD_SIMDJSON OFF CACHE BOOL "Disable implicit fastgltf downloads" FORCE)
    set(FASTGLTF_DOWNLOAD_SIMDUTF OFF CACHE BOOL "Disable implicit fastgltf downloads" FORCE)
    set(FASTGLTF_ENABLE_INSTALL OFF CACHE BOOL "Do not install third-party parser" FORCE)
    set(FASTGLTF_ENABLE_TESTS OFF CACHE BOOL "Do not fetch third-party test dependencies" FORCE)
    set(FASTGLTF_ENABLE_EXAMPLES OFF CACHE BOOL "Do not fetch third-party example dependencies" FORCE)
    set(FASTGLTF_ENABLE_DOCS OFF CACHE BOOL "Do not build third-party docs" FORCE)
    set(FASTGLTF_ENABLE_FUZZING OFF CACHE BOOL "Fuzzing is added at engine boundary separately" FORCE)
    set(FASTGLTF_ENABLE_CPP_MODULES OFF CACHE BOOL "Use the ordinary C++20 target" FORCE)
    add_subdirectory("${_astro_fastgltf_source}" "${CMAKE_BINARY_DIR}/_deps/fastgltf" EXCLUDE_FROM_ALL)

    add_library(astro_gltf GltfImporter.cpp SceneMeshResolver.cpp)
    target_include_directories(astro_gltf PUBLIC include)
    target_link_libraries(astro_gltf PUBLIC astro_runtime_render3d astro_assets astro_scene PRIVATE fastgltf::fastgltf)
    target_compile_features(astro_gltf PUBLIC cxx_std_20)
    set_target_properties(astro_gltf PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(astro_gltf PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
    else()
        target_compile_options(astro_gltf PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
