add_library(astro_runtime_input runtime/InputMapping.cpp)
target_include_directories(astro_runtime_input PUBLIC include)
target_link_libraries(astro_runtime_input PUBLIC astro_core)
target_compile_features(astro_runtime_input PUBLIC cxx_std_20)
set_target_properties(astro_runtime_input PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(astro_runtime_input PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
else()
    target_compile_options(astro_runtime_input PRIVATE -Wall -Wextra -Wpedantic)
endif()
